# Try to enable the LAgentUser service
systemctl --user import-environment  DISPLAY
systemctl --user import-environment  WAYLAND_DISPLAY
systemctl --user start LAgentUser.service > /dev/null
systemctl --user status LAgentUser.service --no-pager > /dev/null
if [ $? -eq 0 ]

fi
